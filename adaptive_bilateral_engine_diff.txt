+void TryCompleteInitialPair()
+{
+   if(!g_initialPairPartial)
+      return;
+
+   if(g_initialRetryCount >= 1)
+   {
+      SetAdaptiveBlockReason("INITIAL_PAIR_PARTIAL", "limite de tentativas da perna ausente");
+      return;
+   }
+
+   datetime now = TimeCurrent();
+   int retry_cooldown = MathMax(5, ReopenDelaySecondsAfterClose);
+   if(g_initialLastAttemptTime > 0 && (int)(now - g_initialLastAttemptTime) < retry_cooldown)
+   {
+      SetAdaptiveBlockReason("INITIAL_PAIR_PARTIAL", "aguardando cooldown da retentativa");
+      return;
+   }
+
+   BuildEntryExposureSnapshot();
+   bool missing_buy = (!g_initialBuyConfirmed && g_exposure.open_buy[(int)ROLE_INITIAL] == 0);
+   bool missing_sell = (!g_initialSellConfirmed && g_exposure.open_sell[(int)ROLE_INITIAL] == 0);
+   if(missing_buy == missing_sell)
+   {
+      g_initialPairPartial = (missing_buy && missing_sell && g_initialPairRequested &&
+                              g_exposure.total > 0);
+      if(!g_initialPairPartial)
+         LogMessage("[INITIAL_PAIR_CONFIRMED] reconciliado cycle=" + IntegerToString(g_basketCycleId));
+      PersistEntryEngineState();
+      return;
+   }
+
+   ENUM_ORDER_TYPE missing_direction = (missing_buy ? ORDER_TYPE_BUY : ORDER_TYPE_SELL);
+   int side_inflight = (missing_buy ? g_exposure.inflight_buy : g_exposure.inflight_sell);
+   if(side_inflight > 0)
+   {
+      SetAdaptiveBlockReason("INITIAL_PAIR_PARTIAL", "perna ausente em transito");
+      return;
+   }
+
+   if(g_dailyLossBlocked || g_dailyProfitBlocked || g_basketDDState == BASKET_DD_HARD)
+   {
+      SetAdaptiveBlockReason("INITIAL_PAIR_PARTIAL", "risco impede completar a perna");
+      return;
+   }
+
+   double lot = NormalizeVolume(FixedLot);
+   if(lot <= 0.0)
+      return;
+
+   g_initialRetryCount++;
+   g_initialLastAttemptTime = now;
+   g_adaptiveEntryAttemptedThisTick = true;
+
+   string missing_comment = (missing_buy ? "IA_FG_INITIAL_BUY" : "IA_FG_INITIAL_SELL");
+   bool retry_ok = (missing_buy ? trade.Buy(lot, _Symbol, 0.0, 0.0, 0.0, missing_comment)
+                                : trade.Sell(lot, _Symbol, 0.0, 0.0, 0.0, missing_comment));
+   uint retry_retcode = (uint)trade.ResultRetcode();
+   if(retry_ok && IsTradeRetcodeSuccess(retry_retcode))
+   {
+      RegisterEntryInFlight(missing_direction);
+      LogMessage("[INITIAL_PAIR_REQUEST] retentativa=" + missing_comment);
+   }
+   else
+      LogMessage("[INITIAL_PAIR_BLOCKED] retentativa_falhou=" + missing_comment +
+                 " retcode=" + IntegerToString((int)retry_retcode) +
+                 " descricao=" + trade.ResultRetcodeDescription());
+
+   InvalidateMoneySnapshots();
+   PersistEntryEngineState();
+}
